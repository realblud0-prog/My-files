local function getVisibleButton(name)
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    local r = pg and pg:FindFirstChild("NoResetGUIHandler", true) and pg.NoResetGUIHandler:FindFirstChild("PropMenuFilter", true)
    if not r then return nil end
    for _, d in ipairs(r:GetDescendants()) do
        if d.Name == name and (d:IsA("ImageButton") or d:IsA("TextButton")) and d.Visible and d.AbsoluteSize.Magnitude > 0 then return d end
    end
    return nil
end

local function forceClick(btn)
    local ok, conns = pcall(getconnections, btn.MouseButton1Click)
    if ok and conns then for _, c in ipairs(conns) do if c.Function then pcall(c.Function) end end end
    pcall(function() btn.MouseButton1Click:Fire() end)
end

local function onJail(target)
    if not target or activeTraps[target] then return end
    task.spawn(function()
        selectedName.Text = "Preparing..."; selectedName.TextColor3 = C.Accent
        local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
        local hum = char:WaitForChild("Humanoid")
        local tool = char:FindFirstChild("PropMaker") or (LocalPlayer:FindFirstChild("Backpack") and LocalPlayer.Backpack:FindFirstChild("PropMaker"))
        if not tool then ReplicatedStorage.RE["1Too1l"]:InvokeServer("PickingTools", "PropMaker"); task.wait(0.5); tool = LocalPlayer.Backpack:FindFirstChild("PropMaker") end
        if not tool then selectedName.Text = "Error: No PropMaker"; return end
        if tool.Parent ~= char then pcall(function() hum:EquipTool(tool) end); task.wait(0.3) end
        local placer = tool:FindFirstChild("Tool_PropMake")
        if not placer then selectedName.Text = "Error: No placer"; return end

        PROP_MENU:FireServer("OpenPropMenu"); task.wait(0.8)
        local cageBtn = getVisibleButton(JAIL_PROP)
        if not cageBtn then local catBtn = getVisibleButton(JAIL_CAT); if catBtn then forceClick(catBtn); task.wait(0.4) end; cageBtn = getVisibleButton(JAIL_PROP) end
        if not cageBtn then selectedName.Text = "Error: BigCage not found"; return end
        
        local tRoot = target.Character and target.Character:FindFirstChild("HumanoidRootPart")
        if not tRoot then selectedName.Text = "Error: Target lost"; return end
        local params = RaycastParams.new(); params.FilterType = Enum.RaycastFilterType.Exclude; params.FilterDescendantsInstances = { char, target.Character }; params.IgnoreWater = true
        local hit = workspace:Raycast(tRoot.Position + Vector3.new(0, 5, 0), Vector3.new(0, -30, 0), params)
        if not hit then selectedName.Text = "Error: No ground"; return end
        
        forceClick(cageBtn); task.wait(0.5); placer:FireServer(hit.Instance, hit.Position)
        local CAGE_CENTER = hit.Position 
        task.wait(0.8)
        
        local knownProps = {}
        local wsCom = workspace:FindFirstChild("WorkspaceCom")
        if wsCom then for _, cat in pairs(wsCom:GetChildren()) do for _, p in pairs(cat:GetChildren()) do if p:FindFirstChild("SetCurrentCFrame") then knownProps[p] = true end end end end
        selectedName.Text = "Drop chair NOW!"; selectedName.TextColor3 = C.Green
        
        local chairRemote = nil
        while not chairRemote do
            task.wait(0.1)
            if wsCom then for _, cat in pairs(wsCom:GetChildren()) do for _, p in pairs(cat:GetChildren()) do if not knownProps[p] and p:FindFirstChild("SetCurrentCFrame") then chairRemote = p.SetCurrentCFrame; break end end; if chairRemote then break end end
        end
        selectedName.Text = "TRAPPED: " .. target.Name; selectedName.TextColor3 = C.Red

        local lastSitTime = 0
        activeTraps[target] = RunService.Heartbeat:Connect(function()
            if not chairRemote or not target or not target.Character then return end
            local tChar = target.Character
            local tRoot = tChar:FindFirstChild("HumanoidRootPart")
            local tHum = tChar:FindFirstChildOfClass("Humanoid")
            if not tRoot or not tHum then return end

            if tHum.Sit then
                if tick() - lastSitTime > 0.3 then
                    lastSitTime = tick()
                    local fPos = CFrame.new(CAGE_CENTER + Vector3.new(0, 1.5, 0))
                    task.spawn(function() pcall(function() chairRemote:InvokeServer(fPos) end) end)
                end
            else
                local fPos = tRoot.CFrame * CFrame.new(0, -3.2, 0)
                task.spawn(function() pcall(function() chairRemote:InvokeServer(fPos) end) end)
            end
        end)
    end)
end

local function onUnjail(target)
    if not target or not activeTraps[target] then return end
    activeTraps[target]:Disconnect(); activeTraps[target] = nil
    selectedName.Text = "Released " .. target.Name; selectedName.TextColor3 = C.Text
    pcall(function() PROP_MENU:FireServer("ClearAllProps") end)
end

                                myVan:PivotTo(voidCFrame)
                                
                                local isVoided = false
                                local voidLoop
                                voidLoop = task.spawn(function()
                                    while not isVoided do
                                        if not target or not target.Character then break end
                                        local currentRoot = target.Character:FindFirstChild("HumanoidRootPart")
                                        if not currentRoot then break end
                                        
                                        local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                                        local isDeep = currentRoot.Position.Y <= -50000
                                        local isFar = myRoot and (currentRoot.Position - myRoot.Position).Magnitude > 20000
                                        
                                        if isDeep or isFar then isVoided = true; break end
                                        
                                        local speed = tick() * 255
                                        local randomSpin = CFrame.Angles(math.sin(speed) * 5, math.cos(speed) * 5, math.tan(speed) * 0.5)
                                        local killPos = currentRoot.CFrame * randomSpin
                                        
                                        task.spawn(function() pcall(function() grabbedRemote:InvokeServer(killPos) end) end)
                                        task.spawn(function() pcall(function() grabbedRemote:InvokeServer(voidCFrame) end) end)
                                        task.wait(0.05)
                                    end
                                    selectedName.Text = "TARGET EXECUTED & VOIDED"; selectedName.TextColor3 = C.Red
                                end)
                            end
                        end)
                    else
                        if tHum.Sit then
                            task.spawn(function() pcall(function() grabbedRemote:InvokeServer(cargoPos) end) end)
                        else
                            local feetPos = CFrame.new(tR.Position - Vector3.new(0, 3.2, 0))
                            task.spawn(function() pcall(function() grabbedRemote:InvokeServer(feetPos) end) end)
                            task.wait(0.03)
                            task.spawn(function() pcall(function() grabbedRemote:InvokeServer(cargoPos) end) end)
                        end
                    end
                end)
            end
        end)
        table.insert(activeVanTraps[target].connections, driveConn)
    end)
end

local function onUnvan(target)
    if not target or not activeVanTraps[target] then return end
    for _, conn in ipairs(activeVanTraps[target].connections) do
        pcall(function() conn:Disconnect() end)
    end
    if activeVanTraps[target].van and activeVanTraps[target].van.Parent then
        activeVanTraps[target].van:Destroy()
    end
    activeVanTraps[target] = nil
    selectedName.Text = "Released " .. target.Name; selectedName.TextColor3 = C.Text
    pcall(function() PROP_MENU:FireServer("ClearAllProps") end)
end

-- KILL / VIEW / TELEPORT / BRING / FLING
-- ==========================================
local function onKill(target)
    if not target or activeTraps[target] then return end
    local remotes = {}
    local wsCom = workspace:FindFirstChild("WorkspaceCom")
    if wsCom then for _, cat in pairs(wsCom:GetChildren()) do for _, p in pairs(cat:GetChildren()) do if p:FindFirstChild("SetCurrentCFrame") then table.insert(remotes, p.SetCurrentCFrame) end end end end
    if #remotes == 0 then selectedName.Text = "Place a prop first!"; selectedName.TextColor3 = C.Red; return end
    selectedName.Text = "EXECUTING: " .. target.Name; selectedName.TextColor3 = C.Red
    local currentIdx = 1
    activeTraps[target] = RunService.Heartbeat:Connect(function()
        if not target or not target.Character then return end
        local tRoot = target.Character:FindFirstChild("HumanoidRootPart")
        if not tRoot then return end
        local remote = remotes[currentIdx]
        if remote then
            local speed = tick() * 255
            local randomSpin = CFrame.Angles(math.sin(speed) * 5, math.cos(speed) * 5, math.tan(speed) * 0.5)
            local killPos = tRoot.CFrame * randomSpin
            task.spawn(function() pcall(function() remote:InvokeServer(killPos) end) end)
        end
        currentIdx = (currentIdx % #remotes) + 1
    end)
end

local function onUnkill(target)
    if not target or not activeTraps[target] then return end
    activeTraps[target]:Disconnect(); activeTraps[target] = nil
    selectedName.Text = "Stopped killing " .. target.Name; selectedName.TextColor3 = C.Text
    pcall(function() PROP_MENU:FireServer("ClearAllProps") end)
end

local function onView(target)
    if target and target.Character and target.Character:FindFirstChild("Humanoid") then
        workspace.CurrentCamera.CameraSubject = target.Character.Humanoid
        selectedName.Text = "Spectating: " .. target.Name; selectedName.TextColor3 = C.Accent
    end
end
local function onUnview()
    if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then workspace.CurrentCamera.CameraSubject = LocalPlayer.Character.Humanoid end
    if selectedPlayer then selectedName.Text = selectedPlayer.Name; selectedName.TextColor3 = C.Text end
end
local function onTeleport(target) if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") and LocalPlayer.Character then LocalPlayer.Character:PivotTo(target.Character.HumanoidRootPart.CFrame * CFrame.new(0, 0, 3)) end end
local function onBring(target) if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") and LocalPlayer.Character then target.Character:PivotTo(LocalPlayer.Character.HumanoidRootPart.CFrame * CFrame.new(0, 0, -3)) end end
local function onFling(target) selectedName.Text = "Flinging " .. target.Name .. "..."; task.delay(2, function() if selectedName.Text == "Flinging " .. target.Name .. "..." then selectedName.Text = target.Name end end) end
local function onUnfling(target) selectedName.Text = target.Name end

local actions = {
    { name = "Jail", activeName = "Unjail", func = onJail, undoFunc = onUnjail },
    { name = "Kidnap", activeName = "Unkidnap", func = onVanTrap, undoFunc = onUnvan },
    { name = "Kill", activeName = "Unkill", func = onKill, undoFunc = onUnkill },
    { name = "View", activeName = "Unview", func = onView, undoFunc = onUnview },
    { name = "Teleport", activeName = "Teleport", func = onTeleport, undoFunc = nil },
    { name = "Bring", activeName = "Bring", func = onBring, undoFunc = nil },
    { name = "Fling", activeName = "Unfling", func = onFling, undoFunc = onUnfling }
}

local actionButtons = {}
for index, action in ipairs(actions) do
    local button = makeButton(action.name, actionGrid, 30)
    button.LayoutOrder = index; actionButtons[action.name] = button
    button.MouseButton1Click:Connect(function()
        if not selectedPlayer then return end
        if not playerStates[selectedPlayer] then playerStates[selectedPlayer] = {} end
        local isActive = playerStates[selectedPlayer][action.name]
        if isActive then
            if action.undoFunc then action.undoFunc(selectedPlayer) end
            playerStates[selectedPlayer][action.name] = false
            button.Text = action.name; button.TextColor3 = C.Text
        else
            action.func(selectedPlayer)
            playerStates[selectedPlayer][action.name] = true
            if action.undoFunc then button.Text = action.activeName; button.TextColor3 = C.Red end
        end
    end)
end

local function selectPlayer(target)
    selectedPlayer = target
    selectedName.Text = target.DisplayName; selectedName.TextColor3 = C.Text
    selectedUsername.Text = "@" .. target.Name
    local success, image = pcall(function() return Players:GetUserThumbnailAsync(target.UserId, Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size100x100) end)
    if success then selectedAvatar.Image = image end
    tween(selectedPanel, 0.15, { BackgroundColor3 = C.Secondary })
    for _, action in ipairs(actions) do
        local button = actionButtons[action.name]
        if playerStates[target] and playerStates[target][action.name] then button.Text = action.activeName; button.TextColor3 = C.Red
        else button.Text = action.name; button.TextColor3 = C.Text end
    end
end

local function refreshPlayers()
    for _, child in ipairs(playerList:GetChildren()) do if child:IsA("TextButton") then child:Destroy() end end
    local query = string.lower(search.Text)
    for _, target in ipairs(Players:GetPlayers()) do
        if target ~= LocalPlayer then
            local display = string.lower(target.DisplayName); local username = string.lower(target.Name)
            if query == "" or string.find(display, query, 1, true) or string.find(username, query, 1, true) then
                local row = create("TextButton", { Size = UDim2.new(1, 0, 0, 28), BackgroundColor3 = C.Secondary, Text = target.DisplayName .. "  @" .. target.Name, TextColor3 = C.Text, TextSize = 9, Font = Enum.Font.GothamMedium, AutoButtonColor = false }, playerList)
                round(row, 8)
                row.MouseEnter:Connect(function() tween(row, 0.12, { BackgroundColor3 = C.Hover }) end)
                row.MouseLeave:Connect(function() tween(row, 0.12, { BackgroundColor3 = C.Secondary }) end)
                row.MouseButton1Click:Connect(function() selectPlayer(target) end)
            end
        end
    end
end

search:GetPropertyChangedSignal("Text"):Connect(refreshPlayers)
refresh.MouseButton1Click:Connect(function() tween(refreshLogo, 0.35, { Rotation = refreshLogo.Rotation + 180 }); refreshPlayers() end)
Players.PlayerAdded:Connect(refreshPlayers)
Players.PlayerRemoving:Connect(function(target)
    if selectedPlayer == target then
        selectedPlayer = nil; selectedName.Text = "No player selected"; selectedUsername.Text = "Waiting..."; selectedAvatar.Image = ""
        for _, action in ipairs(actions) do actionButtons[action.name].Text = action.name; actionButtons[action.name].TextColor3 = C.Text end
    end
    playerStates[target] = nil
    if activeTraps[target] then activeTraps[target]:Disconnect(); activeTraps[target] = nil end
    if activeVanTraps[target] then
        for _, conn in ipairs(activeVanTraps[target].connections) do pcall(function() conn:Disconnect() end) end
        if activeVanTraps[target].van and activeVanTraps[target].van.Parent then activeVanTraps[target].van:Destroy() end
        activeVanTraps[target] = nil
    end
    refreshPlayers()
end)

local dragging, dragStart, startPosition
header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true; dragStart = input.Position; startPosition = main.Position
        input.Changed:Connect(function() if input.UserInputState == Enum.UserInputState.End then dragging = false end end)
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        main.Position = UDim2.new(startPosition.X.Scale, startPosition.X.Offset + delta.X, startPosition.Y.Scale, startPosition.Y.Offset + delta.Y)
    end
end)

-- Resize Handle Logic
local resizing, resizeStart, startSize = false, nil, nil
local resizeHandle = create("TextButton", { Size = UDim2.fromOffset(16, 16), Position = UDim2.new(1, -20, 1, -20), BackgroundColor3 = C.Accent, Text = "◢", TextColor3 = C.Text, TextSize = 10, AutoButtonColor = false, ZIndex = 10 }, main)
round(resizeHandle, 4)

resizeHandle.MouseButton1Down:Connect(function()
    resizing = true
    resizeStart = UserInputService:GetMouseLocation()
    startSize = main.AbsoluteSize
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then resizing = false end
end)

UserInputService.InputChanged:Connect(function(input)
    if resizing and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local current = UserInputService:GetMouseLocation()
        local deltaX = current.X - resizeStart.X
        local deltaY = current.Y - resizeStart.Y
        
        local newWidth = math.max(250, startSize.X + deltaX)
        local newHeight = math.max(250, startSize.Y + deltaY)
        
        main.Size = UDim2.fromOffset(newWidth, newHeight)
        normalSize = UDim2.fromOffset(newWidth, newHeight)
    end
end)
