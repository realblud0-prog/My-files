local function
getVisibleButton(name) local pg =
LocalPlayer:FindFirstChild("PlayerGui") local r = pg and
pg:FindFirstChild("NoResetGUIHandler", true) and
pg.NoResetGUIHandler:FindFirstChild("PropMenuFilter", true) if not r
then return nil end for _, d in ipairs(r:GetDescendants()) do if d.Name
== name and (d:IsA("ImageButton") or d:IsA("TextButton")) and d.Visible
and d.AbsoluteSize.Magnitude > 0 then return d end end return nil end

local function forceClick(btn) local fired = 0 local ok, conns =
pcall(getconnections, btn.MouseButton1Click) if ok and conns then for _,
c in ipairs(conns) do if c.Function then pcall(c.Function); fired =
fired + 1 end end end pcall(function() btn.MouseButton1Click:Fire() end)
return fired > 0 end

local function onJail(target) if not target or activeTraps[target] then
return end task.spawn(function() selectedName.Text = "Preparing…"
selectedName.TextColor3 = C.Accent

        local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
        local hum = char:WaitForChild("Humanoid")

        local tool = char:FindFirstChild("PropMaker") or (LocalPlayer:FindFirstChild("Backpack") and LocalPlayer.Backpack:FindFirstChild("PropMaker"))
        if not tool then
            local getTool = ReplicatedStorage.RE["1Too1l"]
            getTool:InvokeServer("PickingTools", "PropMaker")
            task.wait(0.5)
            tool = LocalPlayer.Backpack:FindFirstChild("PropMaker")
        end
        if not tool then selectedName.Text = "Error: No PropMaker"; return end
        if tool.Parent ~= char then pcall(function() hum:EquipTool(tool) end); task.wait(0.3) end

        local placer = tool:FindFirstChild("Tool_PropMake")
        if not placer then selectedName.Text = "Error: No placer"; return end

        PROP_MENU:FireServer("OpenPropMenu")
        task.wait(0.8)

        local cageBtn = getVisibleButton(JAIL_PROP)
        if not cageBtn then
            local catBtn = getVisibleButton(JAIL_CAT)
            if catBtn then forceClick(catBtn); task.wait(0.4) end
            cageBtn = getVisibleButton(JAIL_PROP)
        end

        if not cageBtn then selectedName.Text = "Error: BigCage not found"; return end

        local tRoot = target.Character and target.Character:FindFirstChild("HumanoidRootPart")
        if not tRoot then selectedName.Text = "Error: Target lost"; return end

        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        params.FilterDescendantsInstances = { char, target.Character }
        params.IgnoreWater = true
        local hit = workspace:Raycast(tRoot.Position + Vector3.new(0, 5, 0), Vector3.new(0, -30, 0), params)

        if not hit then selectedName.Text = "Error: No ground"; return end

        forceClick(cageBtn)
        task.wait(0.5) 
        placer:FireServer(hit.Instance, hit.Position)

        local CAGE_CENTER = hit.Position 

        task.wait(0.8)

        local knownProps = {}
        local wsCom = workspace:FindFirstChild("WorkspaceCom")
        if wsCom then
            for _, cat in pairs(wsCom:GetChildren()) do
                for _, p in pairs(cat:GetChildren()) do
                    if p:FindFirstChild("SetCurrentCFrame") then
                        knownProps[p] = true 
                    end
                end
            end
        end

        selectedName.Text = "Drop chair NOW!"
        selectedName.TextColor3 = C.Green

        local chairRemote = nil

        while not chairRemote do
            task.wait(0.1)
            if wsCom then
                for _, cat in pairs(wsCom:GetChildren()) do
                    for _, p in pairs(cat:GetChildren()) do
                        if not knownProps[p] and p:FindFirstChild("SetCurrentCFrame") then
                            chairRemote = p.SetCurrentCFrame
                            break
                        end
                    end
                    if chairRemote then break end
                end
            end
        end

        selectedName.Text = "TRAPPED: " .. target.Name
        selectedName.TextColor3 = C.Red

        local lastSitTime = 0

        activeTraps[target] = RunService.Heartbeat:Connect(function()
            if not chairRemote then return end
            if not target or not target.Character then return end

            local tChar = target.Character
            local tRoot = tChar:FindFirstChild("HumanoidRootPart")
            local tHum = tChar:FindFirstChildOfClass("Humanoid")
            if not tRoot or not tHum then return end

            if tHum.Sit then
                if tick() - lastSitTime > 0.3 then
                    lastSitTime = tick()
                    local fPos = CFrame.new(CAGE_CENTER + Vector3.new(0, 1.5, 0))
                    task.spawn(function() 
                        pcall(function() chairRemote:InvokeServer(fPos) end) 
                    end)
                end
            else
                local fPos = tRoot.CFrame * CFrame.new(0, -3.2, 0)
                task.spawn(function() 
                    pcall(function() chairRemote:InvokeServer(fPos) end) 
                end)
            end
        end)
    end)

end

local function onUnjail(target) if not target or not activeTraps[target]
then return end activeTraps[target]:Disconnect() activeTraps[target] =
nil selectedName.Text = "Released" .. target.Name
selectedName.TextColor3 = C.Text pcall(function()
PROP_MENU:FireServer("ClearAllProps") end) end

"Flinging" .. target.Name .. "…" then selectedName.Text = target.Name
end end) end local function onUnfling(target) selectedName.Text =
target.Name end local function onKill(target) end local function
onView(target) end local function onTeleport(target) if target and

LocalPlayer.Character:PivotTo(target.Character.HumanoidRootPart.CFrame *

func = onBring, undoFunc = nil }, { name = "Teleport", activeName =
"Teleport", func = onTeleport, undoFunc = nil }, { name = "Kill",
activeName = "Kill", func = onKill, undoFunc = nil }, { name = "View",
activeName = "View", func = onView, undoFunc = nil } }

local actionButtons = {}

for index, action in ipairs(actions) do local button =
makeButton(action.name, actionGrid, 34) button.LayoutOrder = index
actionButtons[action.name] = button

    button.MouseButton1Click:Connect(function()
        if not selectedPlayer then return end
        if not playerStates[selectedPlayer] then playerStates[selectedPlayer] = {} end

        local isActive = playerStates[selectedPlayer][action.name]

        if isActive then
            if action.undoFunc then action.undoFunc(selectedPlayer) end
            playerStates[selectedPlayer][action.name] = false
            button.Text = action.name
            button.TextColor3 = C.Text
        else
            action.func(selectedPlayer)
            playerStates[selectedPlayer][action.name] = true

            if action.undoFunc then 
                button.Text = action.activeName
                button.TextColor3 = C.Red
            end
        end
    end)

end

local function selectPlayer(target) selectedPlayer = target
selectedName.Text = target.DisplayName selectedName.TextColor3 = C.Text
selectedUsername.Text = "@" .. target.Name

    local success, image = pcall(function()
        return Players:GetUserThumbnailAsync(target.UserId, Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size100x100)
    end)
    if success then selectedAvatar.Image = image end

    tween(selectedPanel, 0.15, { BackgroundColor3 = C.Secondary })

    for _, action in ipairs(actions) do
        local button = actionButtons[action.name]
        local isActive = playerStates[target] and playerStates[target][action.name]
        if isActive then
            button.Text = action.activeName
            button.TextColor3 = C.Red
        else
            button.Text = action.name
            button.TextColor3 = C.Text
        end
    end

end
